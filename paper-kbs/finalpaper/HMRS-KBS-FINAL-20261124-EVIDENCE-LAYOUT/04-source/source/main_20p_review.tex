%% Single-column, double-spaced review layout for the same manuscript.
\def\HMRSReviewLayout{}
\AddToHook{class/elsarticle/after}{%
  \RequirePackage{setspace,times,geometry}%
  \geometry{margin=1in}%
}
\AddToHook{package/graphicx/after}{%
  \setkeys{Gin}{height=.60\textheight,keepaspectratio}%
}
\AddToHook{env/frontmatter/after}{\doublespacing}
\AddToHook{env/tabular/begin}{\footnotesize\setstretch{1}}
\makeatletter
%% Let review floats use dedicated pages when the double-spaced text
%% leaves too little room.
\AddToHook{begindocument/before}{%
  \let\hmrs@review@xfloat\@xfloat
  \def\@xfloat#1[#2]{\hmrs@review@xfloat{#1}[#2p]}%
}
\makeatother
