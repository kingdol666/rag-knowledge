\begin{figure*}[t]

\includegraphics[width=0.72\textwidth]{../figures/fig1_rag_vs_kb.png}
\caption{Conventional dense RAG (left) vs.\ the \sys{} protocol (right) on the
same 50-paper corpus and 10 questions: routed category bases, vector-first
recall, an interpretable content gate with librarian fallback, and five-section
cited answers. The bottom strip compares on-target answers, abstentions (the
protocol's two gate-fails were rescued by its librarian), scope-labeled latency
(dense end-to-end; KB retrieval plus reads), and citation granularity; all
numbers are Section~\ref{sec:eval}'s benchmark run.}
\Description{Side-by-side teaser. Left: a three-step dense-RAG stack (fixed
800-character chunking, single dense retrieval pass, 4,000-character evidence
pack to the LLM) with pain chips noting two honest abstentions, missing
section provenance, and no retry. Right: the QDCVR protocol with content-routed
category KBs, query rewrite, vector-first recall, a 0--8 content gate, a
librarian fallback, and a five-section cited answer with gain chips noting
10/10 gold documents hit and zero abstentions. A dark bottom strip compares
the stacks on answered questions (8/10 vs 10/10), unanswered questions, latency
(dense end-to-end 12.2 s vs KB retrieval-plus-reads 2.4 s), and citation
granularity (chunk without path vs base/doc/part/section).}
\label{fig:teaser}
\end{figure*}

\begin{figure*}[t]
\centering
\includegraphics[width=0.72\textwidth]{../figures/fig3_benchmark.png}
\caption{Three answering tracks on the same corpus and questions: Track~A, the
\sys{} protocol; Track~B, a bare agent with whole-file tools and no index;
Track~C, a dense baseline over 800-character fixed chunks; the cards report
each track's outcome verbatim. Bottom strips: per-track mean latencies with
explicit scopes (A excludes answer generation) and BQ01's three verbatim fates;
all numbers are Section~\ref{sec:eval}'s benchmark run (repository artifacts).}
\Description{Three side-by-side track cards over a shared 50-paper corpus and
10 questions. Track A (KB protocol): 10/10 gold documents with per-part source
paths, zero unanswered (two gate-fails rescued by the librarian), verbatim
base/doc/part/section citations with declared blind spots, ~2.4 s retrieval and
reads. Track B (bare agent): 10/10 gold documents but provenance stops at file
names, no section-level trace, 78.8 s average. Track C (dense RAG baseline):
content matches but the harness logs no source paths, 2/10 abstained after the
evidence window missed the passage, 12.2 s average. Below, a latency bar strip
compares 2.4 s, 12.2 s, and 78.8 s with scope labels; a final strip traces
question BQ01 verbatim: Track A answers with part citations, Track B prints the
formula without a link to a section, Track C abstains.}
\label{fig:threeway}
\end{figure*}
