  agent/.style={draw, dashed, rounded corners=2pt, align=center, fill=black!3,

% ── clients ──────────────────────────────────────────────────────
\node[agent, minimum width=26mm] (human) at (0,0)  {Human\\ (web UI)};
\node[agent, minimum width=26mm] (llm)   at (0,-1.9) {LLM agent\\(MCP client)};

% ── service tier ─────────────────────────────────────────────────
\node[svc, minimum width=32mm] (web)  at (4.4,0)    {Nuxt web layer\\ \texttt{:6789}\\ \footnotesize(proxy + KB CRUD)};
\node[svc, minimum width=32mm] (api)  at (9.0,-0.95) {FastAPI backend\\ \texttt{:8771}\\ \footnotesize(parse/search/graph/soul)};
\node[svc, minimum width=30mm] (mcp)  at (4.4,-1.9) {kb-mcp server\\ \footnotesize 94 MCP tools (stdio)};
\node[svc, minimum width=30mm] (mineru) at (13.6,-0.95) {MinerU OCR\\ \footnotesize(ephemeral port)};

% ── storage layers ───────────────────────────────────────────────
\node[layer, minimum width=17mm] (l1) at (2.4,-4.0)  {L1 disk\\ \footnotesize files};
\node[layer, minimum width=17mm] (l2) at (5.0,-4.0)  {L2 tree\\ \footnotesize\texttt{.tree-fs.json}};
\node[layer, minimum width=17mm] (l3) at (7.6,-4.0)  {L3 catalogue\\ \footnotesize\texttt{.kb.yml}};

% ── edges ────────────────────────────────────────────────────────
\draw[ar] (human) -- (web);
\draw[ar] (llm)   -- (mcp);

\draw[ar, very thick] (api.south) -- node[lbl,right,pos=0.35]{single write path} (l4.north);

\draw[ar, densely dotted, thick] (mcp.north east) to[out=20,in=200]

% ── annotation ───────────────────────────────────────────────────
\node[lbl, anchor=west] at (0.1,-5.35)
  {Solid = control flow;\; very thick = the single atomic write path (all five
   layers, one logical transaction);\; dotted = the read path used by retrieval,
   which bypasses the service tier entirely.};
\end{tikzpicture}}
\caption{System architecture. Three services partition responsibility --- the web
layer owns knowledge-base and document CRUD, the FastAPI backend owns parsing,
retrieval, graph and experience computation, and an ephemeral MinerU process does
OCR. Every mutation travels one write path that updates all five storage layers
inside a single logical transaction (\S\ref{sec:sys-layers}); retrieval reads
those layers directly, so search adds no load to the service tier. The same
capability set is published twice: as REST endpoints for the human UI, and as 94
MCP tools for agents, so a procedure validated in one surface is executable in
the other.}
\label{fig:arch}
\end{figure*}
